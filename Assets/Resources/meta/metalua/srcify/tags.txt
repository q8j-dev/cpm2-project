local tags = {}
local astModifiers = require "metalua.srcify.astModifiers"
function tags:Class(cls)
    local classId = cls[1]
    local members = cls[2]
    local classAttributes = cls[3]
    local primaryCtor = cls[4]
    local inheritance = cls[5]
    local staticFields = {}
    local instanceFields = {}
    local staticMethods = {}
    local instanceMethods = {}

    local function getAttr(node, attr)
        local attrsIdx = node.tag == 'MethodDef' and 3 or 2
        for i,p in ipairs(node[attrsIdx]) do
            if p[1] == attr then
                return p
            end
        end
    end

    if primaryCtor then
        local args = {}
        for i, p in ipairs(primaryCtor[1]) do
            table.insert(instanceFields, p)
            table.insert(args, p[1])
        end
        table.insert(members, asth.MethodDef{asth.Id{classId[1]}, asth.Function{args,{}}, asth.AttrList{asth.Attr{'static'}}})
    end

    local ctor = nil
    for i,p in ipairs(members) do
        if p.tag == 'MethodDef' then
            if p[1][1] == classId[1] then
                p[1][1] = '.ctor'
                ctor = p
                table.insert(p[3], asth.Attr{'static'})
            end

            if getAttr(p, 'static') then
                table.insert(staticMethods, p)
            else
                table.insert(instanceMethods, p)
            end
        elseif p.tag == 'FieldDef' then
            if getAttr(p, 'static') then
                table.insert(staticFields, p)
            else
                table.insert(instanceFields, p)
            end
        end
    end
    if not ctor then
        ctor = asth.MethodDef{asth.Id{'.ctor'}, asth.Function{{},{}}, asth.AttrList{asth.Attr{'static'}}}
        table.insert(staticMethods, ctor)
    end

    astModifiers.precompile(classId, staticFields, staticMethods, instanceFields, instanceMethods)

    local staticMembers = {}
    for i,p in ipairs(staticFields) do
        if getAttr(p, 'default') then
            local default = getAttr(p, 'default')
            table.insert(staticMembers, asth.Pair{asth.String{p[1][1]}, default[2][1], p})
        end
    end
    for i,p in ipairs(staticMethods) do
        local method = p[2]
        for _, attr in ipairs(p[3]) do
            if attr[1] ~= 'static' then
                method = asth.Call{asth.Id{attr[1]}, method, unpack(attr[2] or {})}
            end
        end
        table.insert(staticMembers, asth.Pair{asth.String{p[1][1]}, method, p})
    end
    
    local instanceMembers = {}
    local ctorArgs = {}
    for i,p in ipairs(ctor[2][1]) do
        ctorArgs[p[1]] = true
    end
    for i,p in ipairs(instanceFields) do
        local pair = asth.Pair{asth.String{p[1][1]}, nil, p}
        local fieldName = p[1][1]
        if getAttr(p, 'default') then
            local default = getAttr(p, 'default')
            if ctorArgs[fieldName] then
                pair[2] = asth.Op{'or', asth.Id{fieldName}, default[2][1]}
            else
                pair[2] = default[2][1]
            end
        else
            if ctorArgs[fieldName] then
                pair[2] = asth.Id{fieldName}
            else
                pair = nil
            end
        end
        if pair then 
            table.insert(instanceMembers, pair)
        end
    end
    for i,p in ipairs(instanceMethods) do
        local method = p[2]
        table.insert(method[1], 1, asth.Id{'self'})
        for _, attr in ipairs(p[3]) do
            method = asth.Call{asth.Id{attr[1]}, method, unpack(attr[2] or {})}
        end
        table.insert(instanceMembers, asth.Pair{asth.String{p[1][1]}, asth.Index{classId, asth.String{p[1][1]}}})
        table.insert(staticMembers, asth.Pair{asth.String{p[1][1]}, method, p})
    end
    table.insert(ctor[2][2], 1, asth.Local{{asth.Id{'self'}}, {}})
    table.insert(ctor[2][2], 2, asth.Set{{asth.Id{'self'}}, {asth.Table{unpack(instanceMembers)}}})
    table.insert(ctor[2][2], asth.Return{asth.Id{'self'}})

    for i,p in ipairs(classAttributes) do
        assert(astModifiers[p[1]], 'Missing AST modifier: '..p[1])
        astModifiers[p[1]](classId, staticMembers, instanceMembers)
    end

    astModifiers.compile(classId, staticMembers, instanceMembers)

    -- local reflectClassTable = asth.Table{}
    -- table.insert(reflectClassTable, asth.Pair{asth.String{'impl'}, asth.Table{unpack(staticMembers)}})
    -- table.insert(reflectClassTable, asth.Pair{asth.String{'name'}, asth.String{classId[1]}})
    -- self:format(asth.AddToList{{asth.Id{'classesToReflect'}}, {reflectClassTable}})

    self:format(asth.Set{{classId}, {asth.Table{unpack(staticMembers)}}})
end

function tags:Id(id)
    if id[2] then
        return self:format(asth.Invoke{asth.Id{id[1]}, asth.Id{'MakeGeneric'}, asth.MethodArg{id[2]}})
    end
    self:emit(id[1], 'id')
end

function tags:Set(set)
    self:format(asth.List{set[1]})
    self:space()
    self:emit('=', 'set')
    self:space()
    self:format(asth.List{set[2]})
end

function tags:Table(t)
    self:curly(function() 
        if #t > 1 then
            self:format(asth.IndentedBlock{t, function() 
                self:emit(',', 'sep') 
                self:newline()
            end})
        else
            self:format(asth.List{t})
        end
    end)
end
local luaKeywords = {
        ["and"] = true, 
        ["async"] = true, 
        ["break"] = true, ["do"] = true, ["else"] = true, ["elseif"] = true,
        ["end"] = true, ["false"] = true, ["for"] = true, ["function"] = true,
        ["goto"] = true, 
        ["if"] = true,
        ["in"] = true, 
        ["local"] = true,
        ["var"] = true,
        ["nil"] = true, ["not"] = true, ["or"] = true, ["repeat"] = true,
        ["return"] = true, 
        ["static"] = true, 
        
        ["then"] = true, ["true"] = true, ["until"] = true, ["while"] = true,
        ["..."] = true, [".."] = true, ["=="] = true, [">="] = true, ["<="] = true, ["~="] = true,
        ["::"] = true 
    }
function tags:Pair(pair)
    if pair[1].tag == 'String' and (pair[1][1]:match("^%w") or pair[1][1]:match("^_")) and not pair[1][1]:match('%-') and not luaKeywords[pair[1][1]] then
        self:format(asth.Id{pair[1][1]})
    else
        self:square(function() self:format(pair[1]) end)
    end
    self:space()
    self:emit("=", 'set')
    self:space()
    self:format(pair[2])
end

function tags:Localrec(localrec)
    self:keyword('local')
    self:space()
    self:keyword('function')
    self:space()
    self:format(localrec[1][1])
    self:format(localrec[2][1], localrec)
end

function tags:String(s)
    local quote1 = s.xtra and s.xtra[1] or "'"
    local quote2 = s.xtra and s.xtra[2] or "'"
    self:emit(quote1, 'quote')
    local toEmit = s[1]
    if quote1 ~= '[[' then
        toEmit = toEmit:gsub('\n', '\\n')
        toEmit = toEmit:gsub('\t', '\\t')
        toEmit = toEmit:gsub('\\', '\\\\')
        toEmit = toEmit:gsub(quote1, '\\'..quote1)
    end
    self:emit(toEmit, 'string')
    self:emit(quote2, 'quote')
end

function tags:Number(s)
    self:emit(tostring(s[1]), 'number')
end

function tags:Function(fun, parentNode)
    if not (parentNode and parentNode.tag == 'Localrec') then
        self:keyword('function')
        self:space()
    end 
    self:round(function() self:format(asth.List{fun[1]}) end)
    self:format(asth.IndentedBlock{fun[2]})
    self:keyword('end')
end
function tags:Index(index) 
    self:format(index[1])
    if index[2].tag == 'Id' and index[2][2] then
        self:emit('.', 'dot')
        self:format(index[2])
        return
    end
    if index[2].tag =='String' and index[2][1]:match("^%w") then
        self:emit('.', 'dot')
        self:emit(index[2][1], 'index')
    else
        self:square(function() self:format(index[2]) end)
    end
end

function tags:ConstructorExpr(constructor)
    self:format(asth.Call{asth.Index{constructor[1], asth.String{".ctor"}}, unpack(constructor[2])})
end

function tags:Attribute(a)
    self:format(asth.Set{
        {a[1]},
        {a[2]}
    })
end;
function tags:Return(ret)
    self:keyword('return')
    if #ret > 0 then
        self:space()
        self:format(asth.List{ret})
    end
end;

local ops = {
        ['add'] = '+',
        ['sub'] = '-',
        ['mul'] = '*',
        ['div'] = '/',
        ['mod'] = '%',
        ['pow'] = '^',
        ['concat'] = '..',
        ['eq'] = '==',
        ['ne'] = '~=',
        ['lt'] = '<',
        ['le'] = '<=',
        ['gt'] = '>',
        ['ge'] = '>=',
        ['and'] = 'and',
        ['or'] = 'or',
        ['not'] = 'not ',
        ['len'] = '#',
        ['unm'] = '-',
        ['Is'] = '*',
        ['Contains'] = '*',
        ['Pipe'] = function(self, op)
            local forin = asth.Forin{
                {asth.Id{'_i_'}, asth.Id{'_p_'}}, 
                {asth.Call{asth.Id{'ipairs'}, op[2]}}, 
                {
                    
                }
            }
            self:format(asth.Call{asth.Paren{asth.Function{{}, {
                forin
            }}}}) 
        end,
        ['Walrus'] = '*',
        ['CachedClosure'] = function(self, op) 
            self:format(asth.Index{op[2], asth.String{'__captured__'..op[3][1]}})
        end
    }
function tags:Op(op)
    local operation = ops[op[1]]
    assert(operation, op[1])
    if type(operation) == 'function' then
        return operation(self, op)
    end
    if op[3] then
        self:format(op[2])
        self:space()
        self:emit(operation, 'op')
        self:space()
        self:format(op[3]) 
    else
        self:emit(operation, 'op')
        self:format(op[2])
    end
end;

function tags:Invoke(c)
    if c[2][2] then
        return self:format(asth.Call{asth.Index{c[1], c[2]}, c[1], unpack(c[3][1])})
    end
    self:format(c[1])
    self:emit(":", 'colon')
    self:emit(c[2][1], 'method')
    self:format(c[3])
end

function tags:If(ifStmt)
    for i = 1, #ifStmt do
        if i % 2 == 1 then
            if i == #ifStmt then
                self:keyword('else')
                self:format(asth.IndentedBlock{ifStmt[i]})
            else
                if i == 1 then
                    self:keyword('if')
                else
                    self:keyword('elseif')
                end
                self:space()
                self:format(ifStmt[i])
                self:space()
                self:keyword('then')
            end
        else
            self:format(asth.IndentedBlock{ifStmt[i]})
        end
    end
    self:keyword('end')
end

end
function tags:Local(c)
    self:keyword("local")
    self:space()
    self:format(asth.List{c[1]})
    if c[2] and #c[2] > 0 then
        self:space()
        self:emit('=', 'set')
        self:space()
        self:format(asth.List{c[2]})
    end
end

    self:keyword('for')
    self:space()
    self:format(fornum[1])
    self:space()
    self:emit('=', 'set')
    self:space()
    self:format(asth.List{fornum, 2, #fornum - 1})

    self:format(asth.IndentedBlock{fornum[#fornum]})
    self:keyword("end")

function tags:RemoveFromList(rfl)
    local p = asth.Id{'_p_'}
    local i = asth.Id{'_i_'}
    self:format(asth.Forin{{i, p}, {asth.Call{asth.Id{'ipairs'}, rfl[1][1]}}, {asth.If{asth.Op{'eq', p, rfl[2][1]}, {
        asth.Call{asth.Index{asth.Id{'table'}, asth.String{'remove'}}, rfl[1][1], i},
        asth.Break{}}}}})
end

function tags:Match(match)
    local cases = {}
    for i,case in ipairs(match[3]) do
        if case[1][1][1][1] == '_' then
            table.insert(cases, case[3])
        else
            table.insert(cases, asth.Op{'eq', case[1][1][1], match[1][1]})
            table.insert(cases, case[3])
        end
    end
    self:format(asth.If{unpack(cases)})
end
